<label>地址<input id="url"></label><button id="save">保存</button><p id="status" role="status"></p><ul id="bookmarks"></ul>
<script>
async function renderList() {
  const response = await fetch('/api/bookmarks'); const rows = await response.json();
  const list = document.querySelector('#bookmarks'); list.replaceChildren();
  for (const row of rows) { const item = document.createElement('li'); item.textContent = row.url; list.append(item); }
}
document.querySelector('#save').addEventListener('click', async () => {
  const response = await fetch('/api/bookmarks', {method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({url:document.querySelector('#url').value})});
  const result = await response.json(); document.querySelector('#status').textContent = response.ok ? '已保存' : result.error;
  await renderList();
});
renderList();
</script></html>
